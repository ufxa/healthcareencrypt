% Figure 2 -- HE-MedInfer Sequence Diagram (P1-017-compliant redesign)
% Classical UML-style sequence diagram: vertical lifelines (Data Owner
% left, Inference Server right, Audit Agent center as the intermediary),
% color-and-linestyle-coded messages with a left-side legend, boxed local
% computations at each lifeline, dashed horizontal phase separators, and
% "(Network)" annotations on the three messages that actually cross the
% Data-Owner/Inference-Server trust boundary (the Audit Agent is
% co-located with the Data Owner, Sec.~\ref{sec:ka}, so AA<->DO traffic
% is not network-annotated). Only the Data Owner holds $sk$ and performs
% decryption (Sec.~\ref{sec:security}); the Audit Agent hashes and logs
% the still-encrypted result and never decrypts. Reuses the exact
% color/linestyle taxonomy already
% established in Figure 1 (fig1_overview.tex) for cross-figure consistency:
% blue solid = control/key material, green dashed = CKKS encrypted
% payload, orange dash-dot = commitment/audit record.
\resizebox{!}{14cm}{%
\begin{tikzpicture}[
  font=\scriptsize,
  >={Stealth[length=2mm,width=1.6mm]},
  lifeline/.style={draw=black!55, line width=0.6pt},
  actorbox/.style={draw=black!70, fill=gray!10, rounded corners=2pt, align=center,
                   inner xsep=4pt, inner ysep=3pt, font=\small\bfseries},
  procbox/.style={draw=black!65, fill=blue!5, rounded corners=1.5pt, align=left,
                  inner xsep=4pt, inner ysep=2.5pt, font=\scriptsize},
  hepibox/.style={draw=orange!70!black, fill=yellow!12, rounded corners=1.5pt,
                  align=left, inner xsep=4pt, inner ysep=2.5pt, font=\scriptsize,
                  line width=1pt},
  ctrl/.style={-{Stealth[length=2mm]}, draw=blue!65!black, thick},
  payload/.style={-{Stealth[length=2mm]}, draw=green!45!black, dashed, thick},
  audit/.style={-{Stealth[length=2mm]}, draw=orange!70!black, dash dot, thick},
  netnote/.style={font=\tiny\itshape, text=black!55},
  phasesep/.style={draw=black!45, dash pattern=on 3pt off 2pt, line width=0.7pt},
  phaselabel/.style={font=\scriptsize\bfseries, text=black!65},
]

% ============== legend (top-left, in-picture, no overlay) ==============
\node[draw=black!40, fill=white, rounded corners=2pt, inner sep=4pt,
      align=left, font=\tiny, anchor=north west] (legend) at (-4.6,0.3) {
  \textbf{Message legend}\\[2pt]
  \tikz\draw[-{Stealth[length=1.6mm]}, draw=blue!65!black, thick] (0,0) -- (0.5,0);\ Control / key material\\[1pt]
  \tikz\draw[-{Stealth[length=1.6mm]}, draw=green!45!black, dashed, thick] (0,0) -- (0.5,0);\ CKKS encrypted payload\\[1pt]
  \tikz\draw[-{Stealth[length=1.6mm]}, draw=orange!70!black, dash dot, thick] (0,0) -- (0.5,0);\ Commitment / audit record
};

% ============== lifeline x-positions ==============
% DO (+ co-located Key Authority) = left, Audit Agent = center (intermediary),
% Inference Server = right -- matches the client-left/server-right/
% intermediary-center convention.
\def\xDO{0}
\def\xAA{6.0}
\def\xIS{12.0}
\def\ytop{0}
\def\ybot{-21.3}

% ============== actor header boxes ==============
\node[actorbox, minimum width=32mm] (hDO) at (\xDO, \ytop) {Data Owner\\[-1pt]{\scriptsize\normalfont (EA; holds $sk$; co-located KA, Sec.~\ref{sec:ka})}};
\node[actorbox, minimum width=28mm] (hAA) at (\xAA, \ytop) {Audit Agent\\[-1pt]{\scriptsize\normalfont (AA; co-located with DO; never holds $sk$)}};
\node[actorbox, minimum width=32mm] (hIS) at (\xIS, \ytop) {Inference Server\\[-1pt]{\scriptsize\normalfont (IA; never holds $sk$)}};

% ============== lifelines (drawn first, background) ==============
\begin{scope}[on background layer]
\draw[lifeline] (\xDO,\ytop-0.8) -- (\xDO,\ybot);
\draw[lifeline] (\xAA,\ytop-0.8) -- (\xAA,\ybot);
\draw[lifeline] (\xIS,\ytop-0.8) -- (\xIS,\ybot);
% trust-boundary shading: DO and AA are co-located / trusted in the
% evaluated baseline (Sec.~\ref{sec:ka}); IS is the only remote,
% untrusted party -- shown as a shaded band, not just asserted in prose,
% so this figure does not repeat Figure 1's earlier visual-vs-prose gap.
\draw[rounded corners=6pt, draw=green!40!black, dash pattern=on 2pt off 2pt,
      fill=green!2, line width=0.5pt]
  (\xDO-2.1,\ytop-1.1) rectangle (\xAA+2.1,\ybot+0.2);
\node[font=\tiny, text=green!35!black, anchor=north west] at (\xDO-2.0,\ytop-1.2)
  {trusted / co-located, $sk$ held only by DO (baseline config., Sec.~\ref{sec:ka})};
\end{scope}

% ============== Phase 0: Key Provisioning ==============
\node[phaselabel, anchor=west] at (\xDO-2.1,-1.7) {Phase 0: Key Provisioning (once, cached)};
\draw[ctrl] (\xDO,-2.4) -- node[above, midway, font=\tiny] {$pk,\,evk$ [static, cached at IS]}
  node[netnote, below, midway] {(Network)} (\xIS,-2.4);
\draw[phasesep] (\xDO-2.1,-2.9) -- (\xIS+2.1,-2.9);

% ============== Phase 1: Encrypted Submission ==============
\node[phaselabel, anchor=west] at (\xDO-2.1,-3.4) {Phase 1: Encrypted Submission};

\node[procbox, anchor=west, text width=38mm] (norm) at (\xDO+0.3,-4.4)
  {\textbf{Normalize:} $x_{\text{norm}}\!\leftarrow\!\textsc{Normalize}(x,[-1,1])$};
\draw[lifeline] (\xDO,-4.1) -- (\xDO,-4.7);

\node[procbox, anchor=west, text width=42mm] (pack) at (\xDO+0.3,-5.7)
  {\textbf{SIMD-Pack:} $\{x_1,\dots,x_k\}\!\leftarrow\!\textsc{SIMD-Pack}(x_{\text{norm}})$\newline\scriptsize 512 records/ciphertext};
\draw[lifeline] (\xDO,-5.4) -- (\xDO,-6.0);

\node[procbox, anchor=west, text width=42mm] (enc) at (\xDO+0.3,-7.1)
  {\textbf{CKKS-Encrypt:} $\mathbf{ct}\!\leftarrow\!\Enc_{pk}(x_{\text{norm}})$\newline\scriptsize $n=2^{15}$, 128-bit security (Sec.~\ref{sec:ckks-depth})};
\draw[lifeline] (\xDO,-6.8) -- (\xDO,-7.4);

\draw[audit] (\xDO,-8.2) -- node[above, midway, font=\tiny, align=center]
  {$\mathsf{com}=H(\mathbf{ct}\,\|\,\tau_{\text{submit}})$} (\xAA,-8.2);

\draw[payload] (\xDO,-9.0) -- node[above, midway, font=\tiny]
  {$\mathbf{ct}(x)$ [6.858\,MB ciphertext]}
  node[netnote, below, midway] {(Network)} (\xIS,-9.0);

\draw[phasesep] (\xDO-2.1,-9.7) -- (\xIS+2.1,-9.7);

% ============== Phase 2: Homomorphic Inference ==============
\node[phaselabel, anchor=west] at (\xDO-2.1,-10.2) {Phase 2: Homomorphic Inference (no decryption)};

\node[procbox, anchor=east, text width=44mm] (lin) at (\xIS-0.3,-11.3)
  {\textbf{$\forall l=1..L$:} $\mathbf{h}\!\leftarrow\!W_l\odot_{\text{HE}}\mathbf{h}+_{\text{HE}}b_l$\newline\scriptsize homomorphic linear layer (4$\times$)};
\draw[lifeline] (\xIS,-11.0) -- (\xIS,-11.6);

\node[procbox, anchor=east, text width=44mm] (act) at (\xIS-0.3,-12.6)
  {\textbf{if $l<L$:} $\mathbf{h}\!\leftarrow\!\widehat{\sigma}_7(\mathbf{h})$\newline\scriptsize degree-7 Chebyshev approx.\ of ReLU (3$\times$)};
\draw[lifeline] (\xIS,-12.3) -- (\xIS,-12.9);

\draw[payload] (\xIS,-13.7) -- node[above, midway, font=\tiny]
  {$\mathbf{ct}(y)$ [encrypted result]}
  node[netnote, below, midway] {(Network)} (\xAA,-13.7);

\draw[phasesep] (\xDO-2.1,-14.4) -- (\xIS+2.1,-14.4);

% ============== Phase 3: Audit Verification ==============
% AA never holds sk (Sec.~\ref{sec:security}): it hashes and logs the
% still-encrypted ct(y) it received in Phase 2, then forwards ct(y) to
% the co-located Data Owner, who alone decrypts.
\node[phaselabel, anchor=west] at (\xDO-2.1,-14.9) {Phase 3: Audit Verification (Verify function)};

\node[procbox, anchor=west, text width=44mm] (log) at (\xAA+0.3,-15.8)
  {\textbf{Log:} $\textsc{AuditLog.Append}(\mathsf{com}, H(\mathbf{ct}(y)))$\newline\scriptsize hash-chained, tamper-evident; no $sk$ needed to hash a ciphertext};
\draw[lifeline] (\xAA,-15.5) -- (\xAA,-16.1);

\node[hepibox, anchor=west, text width=50mm] (hepi) at (\xAA+0.3,-17.6)
  {\textbf{Novel Metric (\HEPI{}):}\newline
   $S_{\text{sec}}\!=\!\min(1,\lambda/128)$;\;
   $O_{\text{HE}}\!=\!(1\!+\!\log_{10}\!\tfrac{L_{\text{HE}}}{L_{\text{plain}}})^{-1}$;\;
   $Q_{\text{acc}}\!=\!1\!-\!\tfrac{|A_{\text{plain}}-A_{\text{HE}}|}{A_{\text{plain}}}$\newline
   $\HEPI(\hat{\Mod})=\alpha S_{\text{sec}}+\beta O_{\text{HE}}+\gamma Q_{\text{acc}}$};
\draw[lifeline] (\xAA,-16.6) -- (\xAA,-18.2);

\draw[payload] (\xAA,-18.9) -- node[above, midway, font=\tiny, align=center]
  {$\mathbf{ct}(y)$ [forwarded, still encrypted]} (\xDO,-18.9);

\node[procbox, anchor=east, text width=40mm] (dec) at (\xDO-0.3,-19.9)
  {\textbf{Decrypt (DO only):} $\hat{y}\leftarrow\mathsf{Dec}_{sk}(\mathbf{ct}(y))$};
\draw[lifeline] (\xDO,-19.6) -- (\xDO,-20.2);

\draw[audit] (\xAA,-20.9) -- node[above, midway, font=\tiny, align=center]
  {$\HEPI(\hat{\Mod})\!\geq\!\tau_{\HEPI}$ [gate; see Fig.~\ref{fig:algorithm} caption]} (\xDO,-20.9);

\end{tikzpicture}%
