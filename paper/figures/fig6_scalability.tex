% Figure 6 — Inference Latency vs Batch Size (DERIVED, not independently
% measured at each size -- see caption in Artigo14.tex). Anchored to the
% single real measurement at B=512 for each method (Sec. VI), extrapolated
% via the measured architectural property that CKKS cost is flat for
% B<=512 (one ciphertext regardless of slot fill) and scales as
% ceil(B/512) for B>512; plaintext is assumed to scale linearly with B on
% a single CPU thread. Log-y axis: the real B=512 anchors span 534,100 ms
% (HE-MedInfer) down to 1.42 ms (plaintext), a ~5.6-decade range (matches
% the paper's "four to five orders of magnitude" latency-overhead framing).
\begin{tikzpicture}
\begin{axis}[
  width=\columnwidth,
  height=5.5cm,
  xlabel={Batch Size (records)},
  ylabel={Inference Latency (ms), log scale},
  xmode=log, log basis x=2,
  ymode=log, log basis y=10,
  xtick={64,128,256,512,1024,2048,4096},
  xticklabels={64,128,256,512,1024,2048,4096},
  xticklabel style={font=\scriptsize, rotate=30, anchor=east},
  yticklabel style={font=\scriptsize},
  xlabel style={font=\scriptsize},
  ylabel style={font=\scriptsize},
  legend pos=south east,
  legend style={font=\scriptsize},
  grid=major,
  grid style={dashed, gray!30},
]
% HE-MedInfer -- flat at the measured 534,100 ms for B<=512, then
% ceil(B/512) x that value (only ciphertext count increases)
\addplot+[mark=square*, blue, thick]
  coordinates {
    (64,    534100)
    (128,   534100)
    (256,   534100)
    (512,   534100)
    (1024, 1068200)
    (2048, 2136400)
    (4096, 4272800)
  };
\addlegendentry{\framework{} (derived)}

% Lee et al. proxy -- flat at the measured 513,000 ms for B<=512
\addplot+[mark=triangle*, red!70, dashed]
  coordinates {
    (64,    513000)
    (128,   513000)
    (256,   513000)
    (512,   513000)
    (1024, 1026000)
    (2048, 2052000)
    (4096, 4104000)
  };
\addlegendentry{Lee et al.\ proxy (derived)}

% Plaintext MLP -- linear extrapolation from the measured 1.42 ms at B=512
% (single CPU thread)
\addplot+[mark=*, green!60!black, dotted]
  coordinates {
    (64,   0.178)
    (128,  0.355)
    (256,  0.710)
    (512,  1.42)
    (1024, 2.84)
    (2048, 5.68)
    (4096, 11.36)
  };
\addlegendentry{Plaintext MLP (derived)}

\end{axis}
\end{tikzpicture}
